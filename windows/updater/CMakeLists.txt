cmake_minimum_required(VERSION 3.15)
project(plug_updater LANGUAGES C CXX)
if(MSVC)
  # Setup and rollback execute these tools outside the application bundle.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
function(verify_vendor_file name expected)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${name}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Untrusted vendored updater dependency: ${name}")
  endif()
endfunction()
verify_vendor_file(json.hpp 9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6)
verify_vendor_file(monocypher.c 02174117935699d418443c75a558a287deb06ef8cf7c1adced61d9047d2f323d)
verify_vendor_file(monocypher.h fcaf6ed771358bb4f40fba016f6518ae86ec02b1b877d2cc35ad92d3a26fd7b3)
verify_vendor_file(monocypher-ed25519.c 97d581639dfa72be08a6d57deb7d79b736be001cb416819cab196d22559d242b)
verify_vendor_file(monocypher-ed25519.h 3a3035181f991a158d0e1c7567258f0bae8ba0f1f23c5512b4a1db1b3c9730ce)
set(PLUG_FEED_PUBLIC_KEYS "$ENV{AUTO_UPDATE_FEED_PUBLIC_KEY}")
if(NOT PLUG_FEED_PUBLIC_KEYS MATCHES "^[A-Za-z0-9+/=,]*$")
  message(FATAL_ERROR "Invalid feed public key encoding")
endif()
configure_file(trusted_keys.h.in trusted_keys.h @ONLY)

add_library(plug_updater_crypto STATIC third_party/monocypher.c third_party/monocypher-ed25519.c)
add_library(plug_updater_security STATIC win_security.cpp)
target_compile_definitions(plug_updater_security PUBLIC UNICODE _UNICODE NOMINMAX)
target_include_directories(plug_updater_security PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(plug_updater_security PUBLIC plug_updater_crypto advapi32 bcrypt crypt32 shell32 ole32 wintrust wtsapi32)
if(MSVC)
  target_compile_options(plug_updater_security PRIVATE /W4 /EHsc /utf-8)
endif()

enable_testing()
add_executable(plug_updater_contract_test contract_test.cpp)
target_link_libraries(plug_updater_contract_test PRIVATE plug_updater_security)
add_test(NAME plug_updater_contract COMMAND plug_updater_contract_test)
add_test(NAME plug_updater_manifest_vector COMMAND plug_updater_contract_test "${CMAKE_CURRENT_SOURCE_DIR}/../../test/fixtures/updater_manifest_v1.json")

add_executable(plug_update_service service_main.cpp)
add_executable(plug_update_client client_main.cpp)
add_executable(plug_update_worker worker_main.cpp)
foreach(target plug_update_service plug_update_client plug_update_worker)
  target_link_libraries(${target} PRIVATE plug_updater_security)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
  endif()
endforeach()
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  install(TARGETS plug_update_service plug_update_client plug_update_worker
    RUNTIME DESTINATION "${CMAKE_INSTALL_PREFIX}/updater" COMPONENT Runtime)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  add_subdirectory("../update_helper" "legacy_helper")
endif()
