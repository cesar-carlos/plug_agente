cmake_minimum_required(VERSION 3.15)
project(plug_install_check LANGUAGES CXX)
add_executable(plug_install_check main.cpp)
target_compile_features(plug_install_check PRIVATE cxx_std_17)
target_compile_definitions(plug_install_check PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(plug_install_check PRIVATE odbc32)
if(MSVC)
  set_property(TARGET plug_install_check PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(plug_install_check PRIVATE /W4 /WX /EHsc /utf-8)
endif()
install(TARGETS plug_install_check RUNTIME DESTINATION "${CMAKE_INSTALL_PREFIX}" COMPONENT Runtime)
